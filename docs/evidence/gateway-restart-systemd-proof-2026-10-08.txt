#!/usr/bin/env python3
"""Scratch probe: does /restart really come back under systemd? (re-runnable)

Run as the MainPID of a throwaway systemd USER unit, this builds the DEPLOYED
gateway from the live config (stub transport: it must not poll the bot token),
decides whether it is supervised, and — on the FIRST start only — lets the real
/restart handler exit the process. If `Restart=always` brings the unit back, the
second start prints its marker and the mechanism is proven.

Unit file used (throwaway, removed after):

    [Unit]
    Description=SCRATCH probe — cortex-gateway /restart supervision check
    [Service]
    Type=simple
    WorkingDirectory=%h/.hermes-cortex/scripts
    EnvironmentFile=-%h/hermes-cortex/.env
    ExecStart=%h/.hermes-cortex/venv/bin/python %h/.hermes/cache/scratch/restart_probe.py
    Restart=always
    RestartSec=3
"""
import json
import os
import sys
import time
from pathlib import Path

sys.path.insert(0, "~/.hermes-cortex/scripts")
import cortex_gateway.agents as agents      # noqa: E402
import cortex_gateway.daemon as daemon      # noqa: E402
import cortex_gateway.models as models      # noqa: E402
import cortex_gateway.sessions as sessions  # noqa: E402

MARKER = Path("~/.hermes/cache/scratch/restart-probe.marker")


class StubTransport:
    offset = 0

    def parse(self, raw):
        return dict(raw)

    def send(self, envelope):
        print(f"SENT {envelope.get('body')}", flush=True)
        return True


cfg = json.loads(Path("~/.hermes-cortex/gateway-esther0001.yaml").read_text())
backends = agents.build_backends(cfg["backends"], {"bus_url": "", "bus_headers": {},
                                                    "secret": os.environ.get("GATEWAY_SECRET", "")})
gw = daemon.Gateway(transport=StubTransport(), backends=backends, default_agent="pi",
                    allowed_users=None,
                    sessions=sessions.SessionBook(Path("/tmp/restart-probe-sessions.json")),
                    models=models.ModelBook(Path("/tmp/restart-probe-models.json")))

print(f"START pid={os.getpid()} unit={gw._own_unit_name()!r} "
      f"supervised={gw._supervised_by_systemd()}", flush=True)

if MARKER.exists():
    print("SECOND_START_OK — systemd restarted the unit after the /restart exit",
          flush=True)
    time.sleep(15)
    sys.exit(0)

MARKER.write_text("first start\n")
print("FIRST_START — invoking the real /restart handler now", flush=True)
gw._handle_restart(999999)          # replies, then exits(0) via _exit_for_restart
time.sleep(10)
print("HANDLER_DID_NOT_EXIT — /restart is broken", flush=True)
sys.exit(9)
