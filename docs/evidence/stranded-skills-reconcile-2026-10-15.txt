```python
#!/usr/bin/env python3
"""Categorise every line git removes from a landed skill file.

Exact-substring matching flags REWORDING as loss; this compares at phrase
level (numbering stripped, whitespace collapsed, emphasis dropped, trailing
punctuation neutralised) and reports PRESERVED / PARTIAL / ABSENT per line.
Read-only. Run from the repo root."""
import re, subprocess
from pathlib import Path

FILES = ["skills/devops/enforcement-gate-migration/SKILL.md",
         "skills/devops/shared-repo-push-gates/SKILL.md"]
NUM = re.compile(r"^\s*(?:\d+\.|[-*])\s+")
EMPH = re.compile(r"[*`]")

def norm(s):
    s = NUM.sub("", s)
    s = EMPH.sub("", s)
    s = s.replace("\u2014", ": ").replace("\u2013", ": ")
    s = re.sub(r"\s+", " ", s).strip()
    return s.rstrip(".:;,-").strip()

for path in FILES:
    old = subprocess.run(["git", "show", "HEAD:" + path],
                         capture_output=True, text=True).stdout
    new = Path(path).read_text()
    cn = norm(new)
    new_set = set(new.splitlines())
    removed = [l for l in old.splitlines() if l not in new_set and l.strip()]
    print("== " + path)
    print("   removed lines (raw text diff): %d" % len(removed))
    absent = []
    for line in removed:
        n = norm(line)
        if len(n) < 4:
            continue
        if n in cn:
            verdict = "PRESERVED (renumbered/reflowed only)"
        else:
            frags = [f.strip() for f in re.split(r"(?<=[.!?])\s+", n) if len(f.strip()) >= 12]
            gone = [f for f in frags if norm(f) not in cn]
            absent += gone
            verdict = ("PARTIAL" if len(gone) < len(frags) else "ABSENT")
            if gone:
                verdict += " -- not verbatim: " + repr(gone)
        print("   - " + verdict)
        print("       old: " + n[:110])
    print("   fragments not verbatim: %d" % len(absent))
    for g in absent:
        print("       NOT VERBATIM: " + g[:140])
    print()
```

```python
#!/usr/bin/env python3
"""Land deployed skill content into the repo, asserting byte-identity per file."""
import hashlib, shutil
from pathlib import Path
REPO = Path("~/hermes-cortex".replace("~", __import__("os").path.expanduser("~")))
DEPLOYED = Path("~/.hermes".replace("~", __import__("os").path.expanduser("~")))
PAIRS = ['skills/devops/cross-agent-design/SKILL.md', 'skills/devops/enforcement-gate-migration/SKILL.md', 'skills/devops/enforcement-gate-migration/references/multi-harness-evidence-writer.md', 'skills/devops/enforcement-gate-migration/templates/fail-closed-checker.py', 'skills/devops/shared-repo-push-gates/SKILL.md']
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
bad = []
for rel in PAIRS:
    s, d = DEPLOYED / rel, REPO / rel
    d.parent.mkdir(parents=True, exist_ok=True)
    before = sha(s)
    shutil.copy2(s, d)
    ok = sha(s) == sha(d) == before
    print("%s %s  sha256=%s  (%d bytes)" % ("OK  " if ok else "FAIL", rel, sha(d)[:16], s.stat().st_size))
    if not ok:
        bad.append(rel)
print()
print("copied=%d failures=%d" % (len(PAIRS) - len(bad), len(bad)))
raise SystemExit(1 if bad else 0)
```
