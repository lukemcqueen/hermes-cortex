## Patch 1: Fix gbrain sync daemon — stop using `--all`

**File:** `scripts/install-gbrain-sync.sh`

**Problem:** The sync script generated by this file runs `gbrain sync --all --no-pull`. The built-in `default` gbrain source has no `--path` and fails silently when included in `--all`, so non-default sources never actually sync.

**Fix:** Change the sync invocation to use `--all` with `--skip default`, which tells gbrain to skip the built-in source. This preserves the "sync everything" behavior without the silent failure.

```diff
- "$BUN" "$GBRAIN" sync --all --no-pull 2>&1
+ "$BUN" "$GBRAIN" sync --all --no-pull --skip default 2>&1
```

Also update the comment above it:
```

# Polls gbrain sync every 120 seconds
# Uses --skip default because the built-in 'default' source has no --path
```

## New File: `scripts/seed-project-brain.sh`

This is a new utility script that auto-populates a brain directory from a project repo.

Create `scripts/seed-project-brain.sh` with this content:

```bash
#!/usr/bin/env bash
# ─────────────────────────────────────────────────────────────
#  seed-project-brain.sh — Populate gbrain from repo docs
#
#  After installing Hermes Cortex, each project needs content
#  in its brain directory before gbrain can index anything.
#  This script seeds a brain dir from a project repository's
#  documentation files.
#
#  Usage:
#    bash seed-project-brain.sh --project=acme-royalty \
#      --repo=/path/to/acme-royalty
#
#    bash seed-project-brain.sh --project=my-project \
#      --auto  # Auto-detect repo from common locations
#
#    bash seed-project-brain.sh --all  # Seed all defined projects
#    bash seed-project-brain.sh --list # Show what would be seeded
# ─────────────────────────────────────────────────────────────
set -euo pipefail

RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; BOLD='\033[1m'; RESET='\033[0m'

info()  { echo -e "${GREEN}✓${RESET} $*"; }
warn()  { echo -e "${YELLOW}⚠${RESET} $*"; }
error() { echo -e "${RED}✗${RESET} $*"; }

BRAIN_DIR="${HOME}/brain"
GBRAIN_CMD="${HOME}/.bun/bin/gbrain"
AUTO_MODE=false
ALL_MODE=false
LIST_MODE=false
PROJECT=""
REPO_DIR=""

# Parse args
for arg in "$@"; do
  case "$arg" in
    --project=*) PROJECT="${arg#*=}" ;;
    --repo=*)    REPO_DIR="${arg#*=}" ;;
    --auto)      AUTO_MODE=true ;;
    --all)       ALL_MODE=true ;;
    --list)      LIST_MODE=true ;;
    --help|-h)
      echo "Usage: bash seed-project-brain.sh [options]"
      echo ""
      echo "  --project=<name>  Brain source name (must match ~/brain/<name>)"
      echo "  --repo=<path>     Path to the project's git repository"
      echo "  --auto            Auto-detect repo at ~/Developer/AI/<project>"
      echo "  --all             Seed all brain dirs from auto-detected repos"
      echo "  --list            List brain dirs and their status, no changes"
      exit 0
      ;;
  esac
done

# Source config file if it exists
CONFIG_FILE="${HOME}/.hermes-cortex/projects.json"
declare -A PROJECT_REPOS

if [[ -f "$CONFIG_FILE" ]]; then
  python3 -c "
import json, sys
with open('$CONFIG_FILE') as f:
    data = json.load(f)
for entry in data.get('projects', []):
    print(f'{entry[\"name\"]}={entry.get(\"repo\", \"\")}')" 2>/dev/null | while IFS='=' read -r name repo; do
    PROJECT_REPOS["$name"]="$repo"
  done
fi

seed_project() {
  local project="$1"
  local repo="$2"
  local brain_dir="${BRAIN_DIR}/${project}"

  if [[ ! -d "$brain_dir" ]]; then
    warn "Brain dir not found: ${brain_dir}"
    warn "  Create it first: mkdir -p ${brain_dir}"
    return 1
  fi

  if [[ ! -d "$repo" ]]; then
    warn "Repo not found for project '${project}': ${repo}"
    warn "  Use --repo=<path> to specify"
    return 1
  fi

  echo -e "${BOLD}[${project}]${RESET}"
  echo "  Repo: ${repo}"
  echo "  Brain: ${brain_dir}"

  # Files to copy from repo to brain
  local DOC_PATTERNS=(
    "README.md"
    "ARCHITECTURE.md"
    "ARCHITECTURE*.md"
    "CONTRIBUTING.md"
    "docs/*.md"
    "docs/architecture/*.md"
    "docs/decisions/*.md"
    "docs/adr/*.md"
    "docs/rfcs/*.md"
    "docs/design/*.md"
    "docs/specs/*.md"
  )

  local copied=0
  for pattern in "${DOC_PATTERNS[@]}"; do
    while IFS= read -r -d '' file; do
      # Compute relative path within repo
      local rel="${file#$repo/}"
      local dest="${brain_dir}/sources/${rel}"
      mkdir -p "$(dirname "$dest")"
      cp "$file" "$dest"
      copied=$((copied + 1))
    done < <(find "$repo" -maxdepth 3 -path "$repo/$pattern" -type f -print0 2>/dev/null || true)
  done

  if [[ "$copied" -eq 0 ]]; then
    warn "  No documentation files found in ${repo}"
    warn "  Create at least a README.md or ARCHITECTURE.md"
    return 1
  fi

  info "  Copied ${copied} documentation file(s)"

  # Ensure git repo
  if [[ ! -d "${brain_dir}/.git" ]]; then
    git -C "$brain_dir" init 2>/dev/null
    info "  Initialized git repo"
  fi

  # Ensure .gitignore
  if [[ ! -f "${brain_dir}/.gitignore" ]]; then
    cat > "${brain_dir}/.gitignore" <<'GITEOF'
MEMORY.md
USER.md
.env
.env.*
*.pem
*.key
*.cert
.DS_Store
Thumbs.db
GITEOF
    info "  Created .gitignore"
  fi

  # Commit
  git -C "$brain_dir" add -A 2>/dev/null
  git -C "$brain_dir" commit -m "seed: ${project} documentation" 2>/dev/null || \
    info "  (nothing new to commit)"

  # Register gbrain source (if not already)
  if ! "$GBRAIN_CMD" sources list 2>/dev/null | grep -q "^  ${project}[[:space:]]"; then
    "$GBRAIN_CMD" sources add "$project" --path "$brain_dir" --name "$project" 2>/dev/null && \
      info "  Registered gbrain source: ${project}" || \
      warn "  Failed to register gbrain source"
  else
    info "  Already registered as gbrain source"
  fi

  # Sync
  echo -n "  Syncing to gbrain... "
  if "$GBRAIN_CMD" sync --source "$project" 2>/dev/null; then
    echo -e "${GREEN}done${RESET}"
  else
    echo -e "${YELLOW}done (check output)${RESET}"
  fi

  # Report pages
  local pages
  pages=$("$GBRAIN_CMD" sources list 2>/dev/null | grep "^  ${project}[[:space:]]" | awk '{print $3}' | head -1)
  pages=${pages:-0}
  if [[ "$pages" -gt 0 ]]; then
    info "  ${pages} page(s) indexed"
  else
    warn "  0 pages indexed — add content and re-sync"
  fi

  echo ""
  return 0
}

# ── List mode ──────────────────────────────────────────────
if [[ "$LIST_MODE" == "true" ]]; then
  echo -e "${BOLD}━━━ Brain Sources Status ━━━${RESET}"
  echo ""
  for dir in "$BRAIN_DIR"/*/; do
    name=$(basename "$dir")
    pages=$("$GBRAIN_CMD" sources list 2>/dev/null | grep "^  ${name}[[:space:]]" | awk '{print $3}' | head -1)
    pages=${pages:-0}
    repo_path=""
    if [[ -f "$CONFIG_FILE" ]]; then
      repo_path=$(python3 -c "import json; p=json.load(open('$CONFIG_FILE')); print([e.get('repo','') for e in p.get('projects',[]) if e.get('name')=='$name'][0] if any(e.get('name')=='$name' for e in p.get('projects',[])) else '')")
    fi
    if [[ -z "$repo_path" ]] && [[ -d "${HOME}/Developer/AI/${name}" ]]; then
      repo_path="${HOME}/Developer/AI/${name}"
    fi
    echo "  ${name}"
    echo "    Pages: $pages"
    if [[ -n "$repo_path" ]]; then
      echo "    Repo:  $repo_path"
    else
      echo "    Repo:  (not detected)"
    fi
    if [[ "$pages" -eq 0 ]]; then
      echo "    ${YELLOW}⚠  Empty — run seed-project-brain.sh${RESET}"
    fi
    echo ""
  done
  exit 0
fi

# ── All mode ───────────────────────────────────────────────
if [[ "$ALL_MODE" == "true" ]]; then
  echo -e "${BOLD}━━━ Seeding All Brain Sources ━━━${RESET}"
  echo ""
  for dir in "$BRAIN_DIR"/*/; do
    name=$(basename "$dir")
    # Skip special dirs
    [[ "$name" == "default" || "$name" == "shared" ]] && continue

    # Try to find repo
    repo=""
    if [[ -n "${PROJECT_REPOS[$name]:-}" ]]; then
      repo="${PROJECT_REPOS[$name]}"
    elif [[ -d "${HOME}/Developer/AI/${name}" ]]; then
      repo="${HOME}/Developer/AI/${name}"
    elif [[ -d "${HOME}/Developer/${name}" ]]; then
      repo="${HOME}/Developer/${name}"
    elif [[ -d "${HOME}/Projects/${name}" ]]; then
      repo="${HOME}/Projects/${name}"
    fi

    if [[ -z "$repo" ]] || [[ ! -d "$repo" ]]; then
      warn "Skipping ${name} — no repo found"
      continue
    fi

    seed_project "$name" "$repo"
  done
  exit 0
fi

# ── Single project mode ────────────────────────────────────
if [[ -z "$PROJECT" ]]; then
  error "No project specified. Use --project=<name> or --all"
  echo "Run with --help for usage."
  exit 1
fi

# Auto-detect repo
if [[ "$AUTO_MODE" == "true" ]] && [[ -z "$REPO_DIR" ]]; then
  for candidate in "${HOME}/Developer/AI/${PROJECT}" "${HOME}/Developer/${PROJECT}" "${HOME}/Projects/${PROJECT}"; do
    if [[ -d "$candidate" ]]; then
      REPO_DIR="$candidate"
      info "Auto-detected repo: ${REPO_DIR}"
      break
    fi
  done
  if [[ -z "$REPO_DIR" ]] && [[ -f "$CONFIG_FILE" ]]; then
    REPO_DIR=$(python3 -c "
import json
with open('$CONFIG_FILE') as f:
    data = json.load(f)
for entry in data.get('projects', []):
    if entry.get('name') == '$PROJECT':
        print(entry.get('repo', ''))
        break
" 2>/dev/null || echo "")
  fi
fi

if [[ -z "$REPO_DIR" ]]; then
  error "No repo path specified and auto-detection failed"
  echo "  Use --repo=/path/to/project or --auto"
  exit 1
fi

seed_project "$PROJECT" "$REPO_DIR"
```
